set(DW1_VERSION us)

# Game translation units and their unit (main or overlay), generated from upstream config (AD-1).
set(DW1_TU_LIST ${CMAKE_BINARY_DIR}/tu_list_${DW1_VERSION}.tsv)
execute_process(
  COMMAND python3 -I tools/port/units/tu_list.py ${DW1_VERSION} -o ${DW1_TU_LIST}
  WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}

file(GLOB DW1_CONFIG_YAML CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/config/${DW1_VERSION}/*.yaml)

  ${CMAKE_SOURCE_DIR}/mk/version/${DW1_VERSION}.mk ${DW1_CONFIG_YAML}
  ${CMAKE_SOURCE_DIR}/tools/port/units/tu_list.py)

# An overlay TU is compiled through a generated wrapper that puts all its global-storage
# variables (globals, file statics, static locals) in the section dw1ovl_<unit> (AD-6).
function(dw1_overlay_wrapper src unit out)
  file(RELATIVE_PATH rel ${CMAKE_SOURCE_DIR} ${src})
  set(wrapper ${CMAKE_BINARY_DIR}/ovl/${rel})
  file(GENERATE OUTPUT ${wrapper} CONTENT
"/* Generated by CMake (AD-6): global-storage variables of ${rel} go to dw1ovl_${unit}. */
#pragma clang attribute push(__attribute__((section(\"dw1ovl_${unit}\"))), apply_to = variable(is_global))
/* A const global would make the section read-only and clash with the writable ones, so the
 * overlay TU is compiled without const; restoring an unchanged constant is harmless. */
#define const
#include \"${src}\"
#pragma clang attribute pop
")
  set(${out} ${wrapper} PARENT_SCOPE)
endfunction()

file(STRINGS ${DW1_TU_LIST} DW1_TU_ROWS)
set(DW1_GAME_SRC)
set(DW1_OVERLAY_UNITS)
# The overlays port/psyq/overlay.c knows (enum Overlay of include/dw/utils.h).
set(DW1_OVERLAY_NAMES btl std fish evl kar vs mov doo2 dooa trn shop dget trn2 murd endi eab)
foreach(row IN LISTS DW1_TU_ROWS)
  string(REPLACE "\t" ";" fields "${row}")
  list(GET fields 0 path)
  list(GET fields 1 unit)
  list(GET fields 2 status)
  if(NOT status STREQUAL "build")
    continue()
  endif()
  if(unit STREQUAL "main")
    list(APPEND DW1_GAME_SRC ${CMAKE_SOURCE_DIR}/${path})
  else()
    if(NOT unit IN_LIST DW1_OVERLAY_NAMES)
      message(FATAL_ERROR "${path}: overlay '${unit}' is not in port/psyq/overlay.c")
    endif()
    dw1_overlay_wrapper(${CMAKE_SOURCE_DIR}/${path} ${unit} wrapper)
    list(APPEND DW1_GAME_SRC ${wrapper})
    list(APPEND DW1_OVERLAY_UNITS ${unit})
  endif()
endforeach()
list(REMOVE_DUPLICATES DW1_OVERLAY_UNITS)
list(LENGTH DW1_OVERLAY_UNITS DW1_OVERLAY_UNIT_COUNT)

# Text headers (text/<dir>/<name>.yaml -> generated/text/<dir>/<name>.h), as mk/text.mk does.
file(GLOB_RECURSE DW1_TEXT_YAML CONFIGURE_DEPENDS RELATIVE ${CMAKE_SOURCE_DIR}/text ${CMAKE_SOURCE_DIR}/text/*/*.yaml)
set(DW1_TEXT_H)
foreach(y IN LISTS DW1_TEXT_YAML)
  string(REGEX REPLACE "\\.yaml$" ".h" h ${y})
  set(out ${CMAKE_BINARY_DIR}/generated/text/${h})
  add_custom_command(OUTPUT ${out}
    COMMAND python3 -I tools/gen_text.py ${DW1_VERSION} text/${y} ${out}
    DEPENDS ${CMAKE_SOURCE_DIR}/text/${y} ${CMAKE_SOURCE_DIR}/tools/gen_text.py
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  list(APPEND DW1_TEXT_H ${out})
endforeach()

# Overlay load addresses (<ovl>_START) for the DW1_PORT block of src/main/utils2.c (AD-4).
set(DW1_OVERLAY_STARTS_H ${CMAKE_BINARY_DIR}/generated/dw1_overlay_starts.h)
add_custom_command(OUTPUT ${DW1_OVERLAY_STARTS_H}
  COMMAND python3 -I -B tools/port/units/ovl_starts.py config/${DW1_VERSION}/overlay.ld
    ${DW1_OVERLAY_STARTS_H}
  DEPENDS ${CMAKE_SOURCE_DIR}/config/${DW1_VERSION}/overlay.ld
    ${CMAKE_SOURCE_DIR}/tools/port/units/ovl_starts.py
  WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

add_compile_options(-std=gnu99 -ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)

# Native: fixed image address so globals, arena and game stack fall in the OT window (AD-5).
if(NOT EMSCRIPTEN)
  add_compile_options(-fno-pie)
  add_link_options(-no-pie -Wl,-Ttext-segment=0x01000000)
endif()
find_package(Threads REQUIRED)

# Game code: same semantics in every preset (AD-17). Port headers come first (AD-2).
add_library(dw1_game_objs OBJECT ${DW1_GAME_SRC} ${DW1_TEXT_H} ${DW1_OVERLAY_STARTS_H})
target_include_directories(dw1_game_objs PRIVATE
  port/psyq/include external/psyq_headers/mw_lib41/include include ${CMAKE_BINARY_DIR}/generated)
target_compile_definitions(dw1_game_objs PRIVATE
  DW1_PORT LANGUAGE_C VERSION_${DW1_VERSION_MACRO})
target_compile_options(dw1_game_objs PRIVATE
  -fsigned-char -fwrapv -fno-strict-aliasing -fno-common -w
  -include ${CMAKE_SOURCE_DIR}/port/psyq/include/port_game.h)

add_library(dw1_psyq STATIC port/psyq/ps1mem.c port/psyq/overlay.c port/psyq/clock.c
  port/psyq/libetc.c port/psyq/libapi.c)
target_link_libraries(dw1_psyq PUBLIC Threads::Threads)
target_compile_options(dw1_psyq PRIVATE -Wall -Wextra -Werror)

  -Wl,--unresolved-symbols=ignore-all -Wl,--wrap=main)
