# An overlay TU is compiled through a generated wrapper that puts all its global-storage
# variables (globals, file statics, static locals) in the section dw1ovl_<unit> (AD-6).
function(dw1_overlay_wrapper src unit out)
  file(RELATIVE_PATH rel ${CMAKE_SOURCE_DIR} ${src})
  set(wrapper ${CMAKE_BINARY_DIR}/ovl/${rel})
  file(GENERATE OUTPUT ${wrapper} CONTENT
"/* Generated by CMake (AD-6): global-storage variables of ${rel} go to dw1ovl_${unit}. */
#pragma clang attribute push(__attribute__((section(\"dw1ovl_${unit}\"))), apply_to = variable(is_global))
/* A const global would make the section read-only and clash with the writable ones, so the
 * overlay TU is compiled without const; restoring an unchanged constant is harmless. */
#define const
#include \"${src}\"
#pragma clang attribute pop
")
  set(${out} ${wrapper} PARENT_SCOPE)
endfunction()

file(STRINGS ${DW1_TU_LIST} DW1_TU_ROWS)
set(DW1_GAME_SRC)
set(DW1_OVERLAY_UNITS)
# The overlays port/psyq/overlay.c knows (enum Overlay of include/dw/utils.h).
set(DW1_OVERLAY_NAMES btl std fish evl kar vs mov doo2 dooa trn shop dget trn2 murd endi eab)
foreach(row IN LISTS DW1_TU_ROWS)
  string(REPLACE "\t" ";" fields "${row}")
  list(GET fields 0 path)
  list(GET fields 1 unit)
  list(GET fields 2 status)
  if(NOT status STREQUAL "build")
    continue()
  endif()
  if(unit STREQUAL "main")
    list(APPEND DW1_GAME_SRC ${CMAKE_SOURCE_DIR}/${path})
  else()
    if(NOT unit IN_LIST DW1_OVERLAY_NAMES)
      message(FATAL_ERROR "${path}: overlay '${unit}' is not in port/psyq/overlay.c")
    endif()
    dw1_overlay_wrapper(${CMAKE_SOURCE_DIR}/${path} ${unit} wrapper)
    list(APPEND DW1_GAME_SRC ${wrapper})
    list(APPEND DW1_OVERLAY_UNITS ${unit})
  endif()
endforeach()
